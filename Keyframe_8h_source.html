158 floatT evaluate(floatT frame) const
159 {
160 if (keyframes_.empty()) return baseValue_;
161
162 if (frame <= keyframes_.front().frame) return keyframes_.front().value;
163 if (frame >= keyframes_.back().frame) return keyframes_.back().value;
164
165 // The segment between two keys is found by walking to the key before
166 // the frame. The tracks hold few keys, so a plain walk stays cheap.
167 const Keyframe* left = &keyframes_.front();
168 const Keyframe* right = &keyframes_.front();
169 for (std::size_t index = 1; index < keyframes_.size(); ++index)
170 {
171 if (keyframes_[index].frame >= frame)
172 {
173 right = &keyframes_[index];
174 break;
175 }
176 left = &keyframes_[index];
177 }
178
179 if (left->interpolation == Interpolation::Step) return left->value;
180
181 const floatT span = right->frame - left->frame;
182 if (span <= 0) return right->value;
183 const floatT fraction = (frame - left->frame) / span;
184
185 if (left->interpolation == Interpolation::Smooth)
186 {
187 // A smoothstep, so the value eases in and out of the segment.
188 const floatT eased = fraction * fraction * (3.f - 2.f * fraction);
189 return left->value + (right->value - left->value) * eased;
190 }
191
192 return left->value + (right->value - left->value) * fraction;
193 }
194
