179 template <class Archive> void serialize(Archive& ar)
180 {
181 // The archive says which way it goes, because the keyframes are
182 // written out in a form the read has to rebuild. The check is made on
183 // the type of the archive, since the archives of cereal carry it as a
184 // member type rather than as a function.
185 if constexpr (Archive::is_loading::value)
186 {
187 ar(CEREAL_NVP(records_));
188 readRecords_();
189 return;
190 }
191
192 records_.clear();
193 for (const auto& [target, track] : tracks_)
194 records_.push_back(makeRecord_(target));
195 ar(CEREAL_NVP(records_));
196 }
