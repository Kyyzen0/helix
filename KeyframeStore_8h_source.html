143 template <class Archive> void serialize(Archive& ar)
144 {
145 // The archive says which way it goes, because the keyframes are
146 // written out in a form the read has to rebuild. The check is made on
147 // the type of the archive, since the archives of cereal carry it as a
148 // member type rather than as a function.
149 if constexpr (Archive::is_loading::value)
150 {
151 ar(CEREAL_NVP(records_));
152 readRecords_();
153 return;
154 }
155
156 records_.clear();
157 for (const auto& [target, track] : tracks_)
158 records_.push_back(makeRecord_(target));
159 ar(CEREAL_NVP(records_));
160 }
